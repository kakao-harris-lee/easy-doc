# Python(Alembic) 스키마 기준선 지문.
#
# 생성 방법 (추론이 아니라 실행 결과다):
#   1. 빈 pgvector/pgvector:pg16 컨테이너 기동
#   2. DATABASE_URL=... uv run alembic upgrade head   → alembic_version = 0006
#   3. SchemaFingerprint.FINGERPRINT_SQL 과 동일한 질의를 psql 로 실행
#
# 이 파일을 손으로 고치지 않는다. Alembic 리비전이 늘면 위 절차를 다시 돌려 갱신하고,
# 그때 V1__python_schema_baseline.sql 도 함께 고친다.
#
# alembic_version·flyway_schema_history 는 제외한다 (마이그레이션 도구의 장부이고,
# 두 적용 경로에서 한쪽에만 존재하므로 넣으면 절대 일치할 수 없다).
#
# 대조 시점의 encryption_scheme(V2)은 포함되지 않는다 — 이 지문은 V1 적용 직후 상태다.
extension vector
table conversions
table documents
table users
table workspaces
column conversions 1 id uuid NOT NULL default=-
column conversions 10 input_tokens integer NULL default=-
column conversions 11 output_tokens integer NULL default=-
column conversions 12 failure_code character varying(64) NULL default=-
column conversions 13 created_at timestamp with time zone NOT NULL default=now()
column conversions 14 updated_at timestamp with time zone NOT NULL default=now()
column conversions 15 edited_text_encrypted bytea NULL default=-
column conversions 16 reviewed_at timestamp with time zone NULL default=-
column conversions 2 document_id uuid NOT NULL default=-
column conversions 3 status character varying(16) NOT NULL default='pending'::character varying
column conversions 4 easy_text_encrypted bytea NULL default=-
column conversions 5 masked_items_encrypted bytea NULL default=-
column conversions 6 key_version smallint NOT NULL default=1
column conversions 7 missing_placeholders jsonb NOT NULL default='[]'::jsonb
column conversions 8 provider_name character varying(32) NULL default=-
column conversions 9 model character varying(128) NULL default=-
column documents 1 id uuid NOT NULL default=-
column documents 10 workspace_id uuid NOT NULL default=-
column documents 2 user_id uuid NOT NULL default=-
column documents 3 title character varying(255) NOT NULL default=-
column documents 4 source_format character varying(16) NOT NULL default=-
column documents 5 source_text_encrypted bytea NOT NULL default=-
column documents 6 key_version smallint NOT NULL default=1
column documents 7 char_count integer NOT NULL default=-
column documents 8 retention_expires_at timestamp with time zone NOT NULL default=(now() + '30 days'::interval)
column documents 9 created_at timestamp with time zone NOT NULL default=now()
column users 1 id uuid NOT NULL default=-
column users 2 email character varying(255) NOT NULL default=-
column users 3 password_hash character varying(255) NOT NULL default=-
column users 4 created_at timestamp with time zone NOT NULL default=now()
column workspaces 1 id uuid NOT NULL default=-
column workspaces 2 user_id uuid NOT NULL default=-
column workspaces 3 name character varying(50) NOT NULL default=-
column workspaces 4 created_at timestamp with time zone NOT NULL default=now()
constraint conversions ck_conversions_status_valid CHECK (((status)::text = ANY ((ARRAY['pending'::character varying, 'processing'::character varying, 'done'::character varying, 'failed'::character varying])::text[])))
constraint conversions fk_conversions_document_id_documents FOREIGN KEY (document_id) REFERENCES documents(id) ON DELETE CASCADE
constraint conversions pk_conversions PRIMARY KEY (id)
constraint documents fk_documents_user_id_users FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE
constraint documents fk_documents_workspace_id_workspaces FOREIGN KEY (workspace_id) REFERENCES workspaces(id)
constraint documents pk_documents PRIMARY KEY (id)
constraint users ck_users_email_lowercase CHECK (((email)::text = lower((email)::text)))
constraint users pk_users PRIMARY KEY (id)
constraint workspaces fk_workspaces_user_id_users FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE
constraint workspaces pk_workspaces PRIMARY KEY (id)
constraint workspaces uq_workspaces_user_id_name UNIQUE (user_id, name)
index conversions ix_conversions_document_id CREATE INDEX ix_conversions_document_id ON public.conversions USING btree (document_id)
index conversions pk_conversions CREATE UNIQUE INDEX pk_conversions ON public.conversions USING btree (id)
index documents ix_documents_retention_expires_at CREATE INDEX ix_documents_retention_expires_at ON public.documents USING btree (retention_expires_at)
index documents ix_documents_user_id CREATE INDEX ix_documents_user_id ON public.documents USING btree (user_id)
index documents ix_documents_workspace_id CREATE INDEX ix_documents_workspace_id ON public.documents USING btree (workspace_id)
index documents pk_documents CREATE UNIQUE INDEX pk_documents ON public.documents USING btree (id)
index users ix_users_email CREATE UNIQUE INDEX ix_users_email ON public.users USING btree (email)
index users pk_users CREATE UNIQUE INDEX pk_users ON public.users USING btree (id)
index workspaces ix_workspaces_user_id CREATE INDEX ix_workspaces_user_id ON public.workspaces USING btree (user_id)
index workspaces pk_workspaces CREATE UNIQUE INDEX pk_workspaces ON public.workspaces USING btree (id)
index workspaces uq_workspaces_user_id_name CREATE UNIQUE INDEX uq_workspaces_user_id_name ON public.workspaces USING btree (user_id, name)
